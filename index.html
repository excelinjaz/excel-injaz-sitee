Sub اضافةعملحدادة() Dim ws As Worksheet Set ws = ThisWorkbook.Sheets("الحدادة") Dim lastRow As Long lastRow = ws.Cells(ws.Rows.Count, 2).End(xlUp).Row + 1 ' إدخال البيانات ws.Cells(lastRow, 2).Value = InputBox("نوع العمل (بوابة / درابزين / مظلة / هيكل):") ws.Cells(lastRow, 3).Value = InputBox("اسم العميل:") ws.Cells(lastRow, 4).Value = InputBox("الموقع:") ws.Cells(lastRow, 5).Value = InputBox("التاريخ (مثال: 2026-09-09):") ws.Cells(lastRow, 6).Value = InputBox("تقييم الجودة (ممتاز / جيد / متوسط):") ' إدراج الصورة Dim imgPath As Variant imgPath = Application.GetOpenFilename("صور (*.jpg;*.png;*.jpeg),*.jpg;*.png;*.jpeg") If imgPath <> False Then Dim pic As Picture Set pic = ws.Pictures.Insert(imgPath) With pic .ShapeRange.LockAspectRatio = msoTrue .Width = 110 .Height = 90 .Top = ws.Cells(lastRow, 1).Top + 5 .Left = ws.Cells(lastRow, 1).Left + 5 End With Else MsgBox "لم يتم إدراج صورة، يمكنك إضافتها لاحقاً" End If ' تلوين الجودة تلقائي Select Case ws.Cells(lastRow, 6).Value Case "ممتاز" ws.Cells(lastRow, 6).Interior.Color = RGB(0, 176, 80) ' أخضر Case "جيد" ws.Cells(lastRow, 6).Interior.Color = RGB(0, 112, 192) ' أزرق Case "متوسط" ws.Cells(lastRow, 6).Interior.Color = RGB(255, 192, 0) ' برتقالي End Select MsgBox "✅ تمت إضافة العمل بنجاح مع الصورة" End Sub
